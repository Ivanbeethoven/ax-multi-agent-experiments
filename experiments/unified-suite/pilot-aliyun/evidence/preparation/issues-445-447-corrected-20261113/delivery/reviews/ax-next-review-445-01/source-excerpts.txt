30:     fn drop(&mut self) {
31:         let mut fills = self.cache.fills.lock().unwrap();
32:         // The map and this guard are the final owners. Waiting followers also
33:         // own guards, so a non-retained result survives until they consume it.
34:         let fill = self.fill.take().unwrap();

38:             // whose Control budget may have room for exactly one fill.
39:             drop(fill);

87:     async fn fill(&self, epoch: u64, key: [u8; 32]) -> PackedResult<FillGuard<'_, T>> {
88:         loop {
89:             let changed = self.fill_changed.notified();
90:             tokio::pin!(changed);
91:             changed.as_mut().enable();
92:             self.check_epoch(epoch)?;
93:             {
94:                 let mut fills = self.fills.lock().unwrap();
95:                 let key = (epoch, key);
96:                 if let Some(fill) = fills.get(&key) {
97:                     return Ok(FillGuard {
98:                         cache: self,
99:                         key,
100:                         fill: Some(fill.clone()),
101:                     });
102:                 }
103:                 if fills.len() < MAX_FILLS {
104:                     let permit = self.budget.admit(&[(V3BudgetPool::Control, FILL_BYTES)])?;
105:                     let fill = Arc::new(Fill {
106:                         value: tokio::sync::Mutex::new(None),
107:                         _permit: permit,
108:                     });
109:                     fills.insert(key, fill.clone());
110:                     return Ok(FillGuard {
111:                         cache: self,
112:                         key,
113:                         fill: Some(fill),
114:                     });
115:                 }
116:             }
117:             tokio::select! {
118:                 () = changed => {},
119:                 () = self.budget.wait_read_cache_epoch_changed(epoch) => {
120:                     return Err(PackedWireError::ReadViewChanged);
121:                 }
122:             }
123:         }
124:     }

137:     fn prepare(&self, generation: ReadGeneration) -> PackedResult<u64> {
138:         if self.budget.state().closed {
139:             self.clear();
140:             return Err(PackedWireError::ReadViewChanged);
141:         }
142:         let mut state = self.state.lock().unwrap();
143:         if state.generation.is_some_and(|old| old != generation) {
144:             // A changed view fences all retained metadata sharing this mount,
145:             // including a fetch that is still awaiting the old backend body.
146:             self.budget.invalidate_read_caches();
147:         }
148:         let epoch = self.budget.read_cache_epoch();
149:         if state.generation != Some(generation) || state.epoch != epoch {
150:             state.entries.clear();
151:             state.bytes = 0;
152:             state.generation = Some(generation);
153:             state.epoch = epoch;
154:         }
155:         Ok(epoch)
156:     }

158:     fn get(&self, key: &[u8; 32], epoch: u64) -> Option<Arc<T>> {
159:         let mut state = self.state.lock().unwrap();
160:         let current_epoch = self.budget.read_cache_epoch();
161:         if state.epoch != current_epoch {
162:             state.entries.clear();
163:             state.bytes = 0;
164:         }
165:         if state.epoch != epoch || current_epoch != epoch {
166:             return None;
167:         }
168:         state.clock = state.clock.wrapping_add(1);
169:         let clock = state.clock;
170:         let entry = state.entries.get_mut(key)?;
171:         entry.accessed = clock;
172:         Some(entry.value.clone())
173:     }

175:     pub(super) async fn read(
176:         &self,
177:         generation: ReadGeneration,
178:         key: [u8; 32],
179:         weight: impl FnOnce(&T) -> u64,
180:         fetch: impl std::future::Future<Output = PackedResult<Arc<T>>>,
181:     ) -> PackedResult<Arc<T>> {
182:         let epoch = self.prepare(generation)?;
183:         if self.capacity == 0 {
184:             let value = fetch.await?;
185:             self.check_epoch(epoch)?;
186:             return Ok(value);
187:         }
188:         if let Some(value) = self.get(&key, epoch) {
189:             self.check_epoch(epoch)?;
190:             return Ok(value);
191:         }
192:         // Only identical epoch/digest pairs wait on each other. The map is
193:         // bounded, its owners are charged, and no registry lock spans I/O.
194:         let fill = self.fill(epoch, key).await?;
195:         let mut completed = tokio::select! {
196:             completed = fill.fill.as_ref().unwrap().value.lock() => completed,
197:             () = self.budget.wait_read_cache_epoch_changed(epoch) => {
198:                 return Err(PackedWireError::ReadViewChanged);
199:             }
200:         };
201:         self.check_epoch(epoch)?;
202:         if let Some(value) = completed.as_ref() {
203:             return Ok(value.clone());
204:         }
205:         if let Some(value) = self.get(&key, epoch) {
206:             self.check_epoch(epoch)?;
207:             return Ok(value);
208:         }
209:         let value = tokio::select! {
210:             value = fetch => value?,
211:             () = self.budget.wait_read_cache_epoch_changed(epoch) => {
212:                 return Err(PackedWireError::ReadViewChanged);
213:             }
214:         };
215:         let bytes = weight(&value).saturating_add(ENTRY_BYTES);
216:         let mut state = self.state.lock().unwrap();
217:         if state.epoch != epoch || self.budget.read_cache_epoch() != epoch {
218:             return Err(PackedWireError::ReadViewChanged);
219:         }
220:         if bytes <= self.capacity {
221:             while state.bytes.saturating_add(bytes) > self.capacity {
222:                 let Some(oldest) = state
223:                     .entries
224:                     .iter()
225:                     .min_by_key(|(_, entry)| entry.accessed)
226:                     .map(|(key, _)| *key)
227:                 else {
228:                     break;
229:                 };
230:                 let removed = state.entries.remove(&oldest).unwrap();
231:                 state.bytes -= removed.bytes;
232:             }
233:             // Ordinary retention pressure must not fail a valid read. All
234:             // actual allocations remain covered by the decoded value owner.
235:             if let Ok(permit) = self.budget.admit(&[(V3BudgetPool::Metadata, ENTRY_BYTES)]) {
236:                 state.clock = state.clock.wrapping_add(1);
237:                 let accessed = state.clock;
238:                 state.entries.insert(
239:                     key,
240:                     Entry {
241:                         value: value.clone(),
242:                         bytes,
243:                         accessed,
244:                         _permit: permit,
245:                     },
246:                 );
247:                 state.bytes += bytes;
248:             }
249:         }
250:         *completed = Some(value.clone());
251:         Ok(value)
252:     }

254:     fn check_epoch(&self, epoch: u64) -> PackedResult<()> {
255:         if self.budget.read_cache_epoch() != epoch || self.budget.state().closed {
256:             return Err(PackedWireError::ReadViewChanged);
257:         }
258:         Ok(())
259:     }

62:     /// prior inserts. try_write prevents cross-cache lock inversion. Each

65:         let mut caches = self.caches.lock().unwrap();
66:         caches.retain(|rows| rows.strong_count() != 0);
67:         for rows in caches.iter().filter_map(Weak::upgrade) {
68:             if Arc::ptr_eq(&rows, current) {
69:                 continue;

136:         self.stop_admission();
